   573→   573→

   577→   577→          set({ currentUser: user });
   578→   578→          return user;
   579→   579→        }
   580→   580→        return null;
   581→   581→      },
   582→   582→
   583→   583→      logout: () => set({ currentUser: null }),
   584→   584→
   585→   585→      addAuditLog: (log) =>
   586→   586→        set((state) => ({

   591→   591→        })),
   592→   592→
   593→   593→      reset: () => {
   594→   594→        // 重新生成种子数据
   595→   595→        set({ ...initialState, currentUser: null });

   601→   601→      },
   602→   602→    }),

   605→   605→      storage: createJSONStorage(() => {
   606→   606→        if (typeof window !== "undefined") {
   607→   607→          return window.localStorage;

   611→   611→          getItem: () => null,
   612→   612→          setItem: () => {},
   613→   613→          removeItem: () => {},
   614→   614→        } as any;
