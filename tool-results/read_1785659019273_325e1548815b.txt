   509→// Store接口
   510→interface StoreState {
   511→  // 当前用户
   512→  currentUser: User | null;
   513→  // 数据集
   514→  users: User[];
   515→  customers: Customer[];
   516→  suppliers: Supplier[];
   517→  products: Product[];
   518→  categories: ProductCategory[];
   519→  inquiries: Inquiry[];
   520→  quotations: Quotation[];
   521→  orders: Order[];
   522→  purchaseRequests: PurchaseRequest[];
   523→  purchaseOrders: PurchaseOrder[];
   524→  warehouses: Warehouse[];
   525→  inventory: Inventory[];
   526→  stockMovements: StockMovement[];
   527→  shipments: Shipment[];
   528→  receivables: Receivable[];
   529→  payables: Payable[];
   530→  documents: Document[];
   531→  auditLogs: AuditLog[];
   532→  systemSettings: SystemSettings;
   533→  exchangeRates: ExchangeRate[];
   534→  notifications: Notification[];
   535→
   536→  // Actions
   537→  login: (email: string, _password: string) => User | null;
   538→  logout: () => void;
   539→  addAuditLog: (log: Omit<AuditLog, "id" | "created_at">) => void;
   540→  // Reset（开发用）
   541→  reset: () => void;
   542→}
   543→

   545→  currentUser: null as User | null,
   546→  users: seedUsers,

   558→  stockMovements: seedStockMovements,
   559→  shipments: seedShipments,
   560→  receivables: seedReceivables,
   561→  payables: seedPayables,
   562→  documents: seedDocuments,
   563→  auditLogs: seedAuditLogs,
   564→  systemSettings: seedSystemSettings,
   565→  exchangeRates: seedExchangeRates,
   566→  notifications: seedNotifications,

   568→
   569→export const useStore = create<StoreState>()(
   570→  persist(
   571→    (set, get) => ({
   572→      ...initialState,
   573→
   574→      login: (email, _password) => {
   575→        const user = get().users.find((u) => u.email === email);
   576→        if (user) {
   577→          set({ currentUser: user });
   578→          return user;
   579→        }
   580→        return null;
   581→      },
   582→
   583→      logout: () => set({ currentUser: null }),
   584→
   585→      addAuditLog: (log) =>
   586→        set((state) => ({
   587→          auditLogs: [
   588→            { ...log, id: uuidv4(), created_at: today() },
   589→            ...state.auditLogs,
   590→          ],
   591→        })),
   592→
   593→      reset: () => {
   594→        // 重新生成种子数据
   595→        set({ ...initialState, currentUser: null });
   596→        // 强制重新加载以重置localStorage
   597→        if (typeof window !== "undefined") {
   598→          localStorage.removeItem("rca-erp-store");
   599→          window.location.reload();
   600→        }
   601→      },
   602→    }),
   603→    {
   604→      name: "rca-erp-store",
   605→      storage: createJSONStorage(() => {
   606→        if (typeof window !== "undefined") {
   607→          return window.localStorage;
   608→        }
   609→        // SSR-safe fallback: in-memory storage
   610→        return {
   611→          getItem: () => null,
   612→          setItem: () => {},
   613→          removeItem: () => {},
   614→        } as any;
   615→      }),
   616→      version: 1,
   617→      skipHydration: false,
   618→    },
   619→  ),
   620→);
